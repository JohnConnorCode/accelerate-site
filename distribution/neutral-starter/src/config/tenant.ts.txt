    domain: string;

    logoMark: string;
    accentColor: string;
    tagline: string;
    emailFooter: string;

export interface TenantPlaybook {
    key: string;

    industry: string;

    publicBooking: boolean;

export interface TenantConfig {

        vercelProjectUrl: string | null;
        supabaseProjectRef: string | null;
    };
}

export function resolvePlaybook(key?: string | null): TenantPlaybook {
    const configured = tenant.playbooks.find((playbook) => playbook.key === key);
    if (configured)
        return configured;
    if (!key && tenant.playbooks[0])
        return tenant.playbooks[0];
    const resolved = key || "roofing";
    return {
        key: resolved,
        label: resolved.charAt(0).toUpperCase() + resolved.slice(1),
        industry: resolved,
        sourceTag: `${resolved}_qualifier`,
        path: `/${resolved}`,
        nextAction: `Respond to qualified ${resolved} audit request`,
    };
}
export const siteUrl = () => process.env.NEXT_PUBLIC_SITE_URL || tenant.brand.siteUrl;
export const adminEmail = () => process.env.ADMIN_EMAIL || tenant.founder.email;
export const fromEmail = () => process.env.RESEND_FROM_EMAIL || `${tenant.brand.name} <${tenant.founder.email}>`;
export const analyticsDomain = () => process.env.NEXT_PUBLIC_PLAUSIBLE_DOMAIN || tenant.brand.domain;
export function supabaseDashboard(path = ""): string {
    const ref = tenant.external.supabaseProjectRef;
    return ref
        ? `https://supabase.com/dashboard/project/${ref}${path}`
        : "https://supabase.com/dashboard/projects";
}
