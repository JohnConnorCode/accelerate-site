export type DistributionProfile = "branded" | "neutral";
/** This exported installation starts neutral even after a fresh Git clone. */
export function distributionProfile(env: Record<string, string | undefined> = process.env): DistributionProfile {
  const value = env.NEXT_PUBLIC_DISTRIBUTION_PROFILE?.trim();
  if (!value || value === "neutral") return "neutral";
  if (value === "branded") return "branded";
  throw new Error('NEXT_PUBLIC_DISTRIBUTION_PROFILE must be "branded" or "neutral".');
}
